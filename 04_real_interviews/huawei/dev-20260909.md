import sys
input = sys.stdin.readline

def solve():
    power = list(map(int, input().split()))
    k = int(input())
    n = len(power)
    total = 0
    for indices in (range(n), range(n - 1, -1, -1)):
        stack = []
        for i in indices:
            while stack and power[stack[-1]] <= power[i]:
                stack.pop()
            if stack and abs(i - stack[-1]) <= k:
                total += power[i] * abs(i - stack[-1])
            stack.append(i)
    print(total % 1000000007)

### 思路分析

### 算法实现
