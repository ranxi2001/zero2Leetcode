   196|## 题解：中位数贪心
   197|### 题目问题拆解
   198|长度为  的环形  串， 与  各  个，每次可以交换环上相邻的两位，问最少交换多少次能让整圈黑白交错。
   199|这是一道把交换翻译成位移的贪心题：难点不在实现，而在看出每个  该去哪一格由它的序号定死，整道题最后只剩一个整体旋转量要挑。
   200|### 算法实现
   201|先看一次交换做了什么。同色交换等于没换，有意义的只有异色交换，它恰好让一个  越过一个 。所以  之间的先后次序永远不变，总交换次数等于各个  位移量之和。
   202|再看终点。长度  的环只有两种交错形态： 全落在偶数位，或全落在奇数位。次序既然不变，第  个  只能去第  个目标格。

   217|def cost_at(c, x):
   218|    """把 c 里的每个数都挪到同一个偶数点 x 上，返回总挪动量"""
   219|    total = 0
   220|    for v in c:
   221|        total += abs(x - v)

   225|def min_swaps(m, d):
   226|    """求让长度 2m 的环形 01 串变成严格交错所需的最少相邻交换次数"""
   227|    # 相邻交换换不了 0 之间的先后次序，所以只需记下每个 0 现在站在哪
   228|    pos = []
   229|    for i, ch in enumerate(d):
   230|        if ch == '0':
   231|            pos.append(i)
   232|    best = None
   233|    # 长度 2m 的环只有两种交错形态：0 全在偶数位（s=0）或全在奇数位（s=1）
   234|    for s in (0, 1):

   238|        c.sort()
   239|        mid = c[m // 2]
   240|        # 上式对 k 是凸函数，实数最优就在中位数上，取离它最近的偶数点即可
   241|        even = mid - ((mid % 2) + 2) % 2
   242|        for x in (even - 2, even, even + 2):
   243|            cur = cost_at(c, x)
   244|            if best is None or cur < best:
   245|                best = cur
   246|    return best

   328|先找无解的必要条件。设每步长度的平方为 ，对任意整点向量  有 ，于是 。把  条向量相加得

   344|import java.io.BufferedReader;
   345|import java.io.InputStreamReader;
   346|import java.io.PrintWriter;
   347|import java.io.StreamTokenizer;

   349|public class Main {
   350|
   351|    // 把 k 个落点依次写进 out；无解返回 false。坐标最大到 10^9，全程用 long
   352|    private static boolean jumpPoints(long x, long y, int k, PrintWriter out) {
   353|        // 终点就是起点：只能靠来回抵消，奇数步凑不出零向量
   354|        if (x == 0 && y == 0) {
   355|            if (k % 2 == 1) {
   356|                return false;
   357|            }
   358|            out.println("YES");
   359|            // 在 (1,0) 与原点之间来回，两种跳跃长度都是 1
   360|            for (int i = 0; i < k; i++) {
   361|                out.println(i % 2 == 0 ? "1 0" : "0 0");
   362|            }
   363|            return true;
   364|        }
   365|
   366|        // 终点非原点且步数为奇数：终点与原点之间来回，最后一步正好停在终点
   367|        if (k % 2 == 1) {
   368|            out.println("YES");
   369|            for (int i = 0; i < k; i++) {
   370|                if (i % 2 == 0) {
   371|                    out.println(x + " " + y);
   372|                } else {
   373|                    out.println("0 0");
   374|                }
   375|            }
   376|            return true;
   377|        }
   378|
   379|        // 步数为偶数时，x+y 必须是偶数，理由见题解里的奇偶不变量
   380|        if (((x + y) % 2 + 2) % 2 != 0) {
   381|            return false;
   382|        }
   383|
   384|        // 把终点拆成等长的两段 A 与 B，两个分量同奇偶时才存在整点拆法
   385|        long ax;
   386|        long ay;

   389|            ay = y / 2;

   392|            ax = (x + y) / 2;
   393|            ay = (y - x) / 2;
   394|        }

   396|        long bx = x - ax;
   397|        long by = y - ay;
   398|
   399|        out.println("YES");

   401|        out.println(x + " " + y);
   402|        // 剩下的偶数步两两一组：退回 B 再走回终点，用的还是同一个长度
   403|        for (int i = 0; i < k / 2 - 1; i++) {
   404|            out.println(bx + " " + by);
   405|            out.println(x + " " + y);
   406|        }
   407|        return true;
   408|    }

   422|            in.nextToken();
