import sys
input = sys.stdin.readline

def solve():
    n = int(input())
    cnt = [0] * (n + 1)
    for _ in range(n - 1):
        u, v = map(int, input().split())
        cnt[min(u, v)] += 1
    acc = 0
    ans = []
    for i in range(1, n + 1):
        acc += cnt[i]
        ans.append(str(1 - i + acc))
    print(' '.join(ans))
